@lru_cache(maxsize=1)
def _tfidf_index():
    """Build a TF‑IDF index over the corpus texts for fallback search."""
    import pandas as pd

    # Keep path logic consistent with the rest of the file.
    # Prefer the same CSV that the app uses.
    corpus_path = os.getenv("CORPUS_PATH", "data/corpus.csv")
    df = pd.read_csv(corpus_path)

    # Require a text column.
    if "text" not in df.columns:
        raise ValueError("Corpus CSV must include a 'text' column for TF‑IDF fallback.")

    texts = df["text"].fillna("").astype(str).tolist()
    vec = TfidfVectorizer(stop_words="english")
    X = vec.fit_transform(texts)
    return df, vec, X


def _search_tfidf(query: str, k: int = 10):
    """Local fallback: TF‑IDF cosine similarity against corpus texts."""
    import pandas as pd

    df, vec, X = _tfidf_index()
    q = (query or "").strip()
    if not q:
        # Return the first k rows if query is empty
        out = df.head(k).copy()
        out["score"] = 0.0
        return out

    qv = vec.transform([q])
    sims = cosine_similarity(qv, X).ravel()

    top_idx = sims.argsort()[::-1][:k]
    out = df.iloc[top_idx].copy()
    out["score"] = sims[top_idx]

    # If your UI relies on these columns, ensure they exist
    for col in ["id", "stage", "spoke", "act", "text", "source", "workstream", "psych_axis"]:
        if col not in out.columns:
            # don't fail; just leave missing columns absent
            pass

    # Sort by score descending for display consistency
    out = out.sort_values("score", ascending=False)
    return out.reset_index(drop=True)
